import java.util.*;

class Solution {
    public List<Integer> findAnagrams(String s, String p) {
        
        List<Integer> result = new ArrayList<>();
        
        if (s.length() < p.length()) {
            return result;
        }

        int[] countS = new int[26];
        int[] countP = new int[26];

        // Frequency of p
        for (char c : p.toCharArray()) {
            countP[c - 'a']++;
        }

        int k = p.length();

        // First window
        for (int i = 0; i < k; i++) {
            countS[s.charAt(i) - 'a']++;
        }

        // Check first window
        if (Arrays.equals(countS, countP)) {
            result.add(0);
        }

        // Sliding window
        for (int i = k; i < s.length(); i++) {
            
            // Add new character
            countS[s.charAt(i) - 'a']++;

            // Remove old character
            countS[s.charAt(i - k) - 'a']--;

            // Check anagram
            if (Arrays.equals(countS, countP)) {
                result.add(i - k + 1);
            }
        }

        return result;
    }
}

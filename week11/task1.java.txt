// Week 11 - Task 1: Majority Element
import java.util.*;

public class Week11Task1 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int candidate = 0;
        int count = 0;

        for (int i = 0; i < n; i++) {
            int value = sc.nextInt();
            if (count == 0) {
                candidate = value;
            }
            count += value == candidate ? 1 : -1;
        }

        System.out.println(candidate);
    }
}

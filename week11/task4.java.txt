// Week 11 - Task 4: Students With the Second Lowest Grade
import java.util.*;

public class Week11Task4 {
    static class Student {
        String name;
        double grade;

        Student(String name, double grade) {
            this.name = name;
            this.grade = grade;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = Integer.parseInt(sc.nextLine().trim());
        List<Student> students = new ArrayList<>();
        TreeSet<Double> grades = new TreeSet<>();

        for (int i = 0; i < n; i++) {
            String name = sc.nextLine().trim();
            double grade = Double.parseDouble(sc.nextLine().trim());
            students.add(new Student(name, grade));
            grades.add(grade);
        }

        Iterator<Double> it = grades.iterator();
        it.next();
        double secondLowest = it.next();

        students.stream()
                .filter(s -> Double.compare(s.grade, secondLowest) == 0)
                .map(s -> s.name)
                .sorted()
                .forEach(System.out::println);
    }
}
